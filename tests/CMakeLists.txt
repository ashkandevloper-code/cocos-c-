# Host-only build of the engine-independent core (game rules, bots, physics, backend protocol, text) + unit tests.
# Usage: cmake -S tests -B build-host -G Ninja && cmake --build build-host && ./build-host/nx_tests
cmake_minimum_required(VERSION 3.10)
project(nexora_core_tests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(NX_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
include(${NX_ROOT}/cmake/nx_core_sources.cmake)
add_library(nx_core STATIC ${NX_CORE_SOURCES})
target_include_directories(nx_core PUBLIC ${NX_ROOT}/Classes ${NX_ROOT}/third_party ${CMAKE_CURRENT_BINARY_DIR}/generated)
nx_generate_config(${CMAKE_CURRENT_BINARY_DIR}/generated)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  # No FMA contraction: physics must match the Dart VM bit-for-bit (online replays + golden tests).
  target_compile_options(nx_core PUBLIC -ffp-contract=off)
  target_compile_options(nx_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-deprecated-declarations -Wno-class-memaccess)
endif()
file(GLOB NX_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(nx_tests ${NX_TEST_SOURCES})
target_link_libraries(nx_tests nx_core)
target_compile_definitions(nx_tests PRIVATE NX_GOLDEN_DIR="${CMAKE_CURRENT_SOURCE_DIR}/golden" NX_RES_DIR="${NX_ROOT}/Resources")
enable_testing()
add_test(NAME nx_tests COMMAND nx_tests)
